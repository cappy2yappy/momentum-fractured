func scan(path):
	for file in DirAccess.get_files_at(path):
		if path.begins_with("res://build") or path.begins_with("res://tests") or path.begins_with("res://docs") or path.begins_with("res://tools"): failures+=1
	for folder in DirAccess.get_directories_at(path):
		if folder!=".godot": scan(path.path_join(folder))
