extends SceneTree
func _init(): call_deferred("run")
func run():
	await process_frame
	var w=load("res://scenes/main.tscn").instantiate();root.add_child(w);current_scene=w
	print("CONTINUOUS START room=",w.room," position=",w.player.position," seeded_abilities=false teleports=false")
	var last_room=-1
	for tick in 6000:
		if w.room!=last_room:
			print("ROUTE room=",w.room," tick=",tick," hp=",w.player.hp);last_room=w.room
		if w.room==2:
			print("ROUTE reached Cistern after earned Windstep: ",w.player.double_jump," guard=",w.guard_defeated)
			break
		Input.action_release("right");Input.action_release("left")
		if w.room==0:
			Input.action_press("right")
			if w.player.position.distance_to(w.right_exit_position())<80: w.interact()
		elif w.room==1:
			if not w.guard_defeated and w.player.position.x>900:
				if is_instance_valid(w.guard):
					var dx=w.guard.position.x-w.player.position.x
					w.player.facing=1 if dx>0 else -1
					if absf(dx)>95: Input.action_press("right" if dx>0 else "left")
				if tick%24==0: w.player.press("attack")
				if tick%60==0: w.player.press("parry")
			else:
				Input.action_press("right")
				if w.player.position.distance_to(w.right_exit_position())<80: w.interact()
		await physics_frame
	Input.action_release("right");Input.action_release("left")
	print("CONTINUOUS END room=",w.room," position=",w.player.position," guard=",w.guard_defeated," windstep=",w.player.double_jump," tether=",w.player.tether_unlocked," dash=",w.player.dash_unlocked)
	w.queue_free();current_scene=null
	for tick in 3: await process_frame
	quit()
